function showSection(sectionId) {
  document.querySelectorAll('.section').forEach(s => s.classList.remove('active'));
  document.querySelectorAll('.nav-btn').forEach(b => b.classList.remove('active'));

  const section = document.getElementById(sectionId);
  if (section) section.classList.add('active');

  const btn = document.querySelector(`[data-section="${sectionId}"]`);
  if (btn) btn.classList.add('active');

  if (sectionId === 'campPicks') {
    loadPickWeeks();
    loadEliteWeekPicks();
  }
  if (sectionId === 'leaderboard' && !APP_DATA) loadAppData();
  if (sectionId === 'seasonTable') loadSeasonResponsesTable();
  if (sectionId === 'bonusLog') loadBonusLog();
  window.scrollTo({ top: 0, behavior: 'smooth' });
}

function loadAppData() {
  google.script.run
    .withSuccessHandler(data => {
      APP_DATA = data;
      renderApp(data);
      loadSeasonResponsesTable();
      loadBonusLog();
    })
    .withFailureHandler(err => showModal('Camp Error', getErrorMessage(err)))
    .getAppData();
}

function renderContentBlocks(blocks) {
  const camp = document.getElementById('campAnnouncementsList');
  camp.innerHTML = '';
  uniqueBy_((blocks.campAnnouncements || []), text => stripHtml(text).toLowerCase()).forEach(text => {
    const div = document.createElement('div');
    div.className = 'announcement';
    div.innerHTML = text;
    camp.appendChild(div);
  });

  const glance = document.getElementById('atAGlanceList');
  glance.innerHTML = '';
  uniqueBy_((blocks.atAGlance || []), text => stripHtml(text).toLowerCase()).forEach(text => {
    const li = document.createElement('li');
    li.innerHTML = text;
    glance.appendChild(li);
  });
}

function renderQuestions(questions) {
  const container = document.getElementById('dynamicQuestions');
  container.innerHTML = '';

  (questions || []).forEach(q => {
    const block = document.createElement('div');
    block.className = 'question-block';

    const title = document.createElement('div');
    title.className = 'question-title';
    title.innerHTML = q.required ? `${q.prompt} *` : q.prompt;
    block.appendChild(title);

    if (q.type === 'text') {
      const textarea = document.createElement('textarea');
      textarea.id = `${q.key}Text`;
      textarea.className = 'text-area';
      textarea.placeholder = 'Share your campfire thoughts';
      textarea.required = !!q.required;
      block.appendChild(textarea);
      container.appendChild(block);
      return;
    }

    const list = document.createElement('div');
    list.className = 'option-grid';

    (q.options || []).forEach(opt => {
      const label = document.createElement('label');
      label.className = 'option-card';

      const input = document.createElement('input');
      input.type = 'radio';
      input.name = q.key;
      input.value = opt.value;
      input.required = !!q.required;

      const media = document.createElement('div');
      media.className = 'option-media';
      if (opt.photoUrl) {
        media.innerHTML = `<img src="${escapeAttr(opt.photoUrl)}" alt="${escapeAttr(opt.label)}">`;
      } else {
        media.innerHTML = `<div class="option-fallback">🏝️</div>`;
      }

      const text = document.createElement('div');
      text.className = 'option-text';
      text.innerHTML = `
        <div class="option-name">${escapeHtml(opt.label || opt.value)}</div>
        ${opt.team ? `<div class="option-meta">${escapeHtml(opt.team)}</div>` : ''}
      `;

      label.appendChild(input);
      label.appendChild(media);
      label.appendChild(text);
      list.appendChild(label);
    });

    block.appendChild(list);
    container.appendChild(block);
  });
}

function renderLeaderboard(data) {
  const body = document.getElementById('leaderboardBody');
  body.innerHTML = '';

  if (!(data.rows || []).length) {
    const tr = document.createElement('tr');
    tr.innerHTML = '<td colspan="4">Leaderboard will appear once players or scores are available.</td>';
    body.appendChild(tr);
    return;
  }

  (data.rows || []).forEach(row => {
    const tr = document.createElement('tr');
    tr.innerHTML = `
      <td>${escapeHtml(String(row.rank || ''))}</td>
      <td><strong>${escapeHtml(row.name || '')}</strong></td>
      <td>${escapeHtml(String(row.weekPoints ?? ''))}</td>
      <td>${escapeHtml(String(row.total ?? ''))}</td>
    `;
    body.appendChild(tr);
  });
}

function renderRecap(updates) {
  const tabs = document.getElementById('recapWeekTabs');
  const wrap = document.getElementById('recapBody');
  if (!tabs || !wrap) return;

  updates = updates || [];
  tabs.innerHTML = '';
  wrap.innerHTML = '';

  if (!updates.length) {
    document.getElementById('recapTitle').textContent = 'Weekly Recap';
    wrap.innerHTML = '<div class="announcement">Weekly recap coming soon.</div>';
    return;
  }

  if (!SELECTED_RECAP_WEEK || !updates.some(update => Number(update.week) === Number(SELECTED_RECAP_WEEK))) {
    SELECTED_RECAP_WEEK = updates[0].week;
  }

  updates.forEach(update => {
    const btn = document.createElement('button');
    btn.type = 'button';
    btn.className = `album-tab ${Number(update.week) === Number(SELECTED_RECAP_WEEK) ? 'active' : ''}`;
    btn.textContent = `Week ${update.week}`;
    btn.onclick = () => {
      SELECTED_RECAP_WEEK = update.week;
      renderRecap(updates);
    };
    tabs.appendChild(btn);
  });

  updates.forEach(update => {
    const active = Number(update.week) === Number(SELECTED_RECAP_WEEK);
    const section = document.createElement('section');
    section.className = `recap-update album-week ${active ? 'active' : ''}`;
    section.innerHTML = `
      <h3 class="section-title mini">${escapeHtml(update.title || `Week ${update.week} Recap`)}</h3>
      <div class="recap-box">${update.body || '<p>Weekly recap coming soon.</p>'}</div>
      <div class="recap-photos">
        <div class="album-week-heading">
          <h3 class="section-title mini">Captured Moments📸</h3>
          <span class="album-count">${escapeHtml(String((update.photos || []).length))} photos</span>
        </div>
        <div class="photo-grid"></div>
      </div>
    `;

    const grid = section.querySelector('.photo-grid');
    if ((update.photos || []).length) {
      (update.photos || []).forEach(photo => grid.appendChild(createPhotoCard_(photo)));
    } else {
      grid.innerHTML = '<div class="announcement">No photos published for this week yet.</div>';
    }
    wrap.appendChild(section);
  });
}

function renderCastGrid(castRows) {
  const wrap = document.getElementById('castGrid');
  if (!wrap) return;
  wrap.innerHTML = '';

  if (!castRows.length) {
    wrap.innerHTML = '<div class="announcement">Cast photos will appear here once the CastStatus sheet is filled in.</div>';
    return;
  }

  uniqueBy_(castRows, player => normalizeTextKey_(player.name)).forEach(player => {
    const status = normalizeCastStatus(player.status);
    const card = document.createElement('button');
    card.type = 'button';
    card.className = `cast-card ${status}`;
    card.onclick = () => showCastCard(player);
    card.innerHTML = `
      <div class="cast-photo-wrap">
        ${player.photoUrl
          ? `<img src="${escapeAttr(player.photoUrl)}" alt="${escapeAttr(player.name)}">`
          : '<div class="cast-photo-fallback">No Photo</div>'}
        <span class="cast-status">${escapeHtml(player.statusLabel || status.toUpperCase())}</span>
      </div>
      <div class="cast-name">${escapeHtml(player.name || '')}</div>
      ${player.team ? `<div class="cast-team">${escapeHtml(player.team)}</div>` : ''}
    `;
    wrap.appendChild(card);
  });
}

function showCastCard(player) {
  const status = player.statusLabel || normalizeCastStatus(player.status).toUpperCase();
  document.getElementById('modalTitle').textContent = player.name || 'Castaway';
  document.getElementById('modalBody').innerHTML = `

    <strong>Status:</strong> ${escapeHtml(status)}<br>
    ${player.team ? `<strong>Tribe:</strong> ${escapeHtml(player.team)}` : ''}
  `;
  document.getElementById('parchmentModal').classList.remove('hidden');
}

function normalizeCastStatus(status) {
  return String(status || 'active').toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '') || 'active';
}

    <div class="photo-caption"><strong>${escapeHtml(photo.name || '')}</strong>${photo.caption ? ` — ${escapeHtml(photo.caption)}` : ''}</div>

function uniqueBy_(items, keyFn) {
  const seen = new Set();
  return (items || []).filter(item => {
    const key = String(keyFn(item) || '').trim();
    if (!key) return true;
    if (seen.has(key)) return false;
    seen.add(key);
    return true;
  });
}

function normalizeTextKey_(value) {
  return String(value || '').trim().toLowerCase().replace(/\s+/g, ' ');
}

function normalizeCampRulesTitle_(title) {
  const safeTitle = String(title || '').trim();
  const plain = stripHtml(safeTitle).trim().toLowerCase();
  return !plain || plain === 'at a glance' ? 'Camp Rules' : safeTitle;
}

function normalizePhotoKey_(photo) {
  return [
    normalizeTextKey_(photo && photo.photoUrl),
    normalizeTextKey_(photo && photo.caption),
    normalizeTextKey_(photo && photo.name)
  ].join('|');
}

function getSelectedValue(name) {
  const checked = document.querySelector(`input[name="${name}"]:checked`);
  return checked ? checked.value : '';
}

function handleSubmit(event) {
  event.preventDefault();

  if (!APP_DATA) {
    showModal('Camp Error', 'App data is still loading.');
    return;
  }

  const name = document.getElementById('playerName').value.trim();
  const email = document.getElementById('playerEmail').value.trim();

  if (!name) {
    showModal('Missing Name', 'Please enter your player name.');
    return;
  }

  const payload = {
    week: APP_DATA.weekNumber,
    name,
    email
  };

  for (const q of (APP_DATA.questions || [])) {
    const plainPrompt = stripHtml(q.prompt);
    if (q.type === 'text') {
      const textValue = document.getElementById(`${q.key}Text`)?.value.trim() || '';
      if (q.required && !textValue) {
        showModal('Incomplete Parchment', `Please answer: ${plainPrompt}`);
        return;
      }
      payload[`${q.key}Label`] = plainPrompt;
      payload[`${q.key}Text`] = textValue;
      payload.commentLabel = plainPrompt;
      payload.commentText = payload[`${q.key}Text`];
    } else {
      const val = getSelectedValue(q.key);
      if (q.required && !val) {
        showModal('Incomplete Parchment', `Please answer: ${plainPrompt}`);
        return;
      }
      payload[`${q.key}Label`] = plainPrompt;
      payload[`${q.key}Pick`] = val;
    }
  }

  localStorage.setItem('survivor51_name', name);
  localStorage.setItem('survivor51_email', email);

  google.script.run
    .withSuccessHandler(r => {
      document.getElementById('savedNameDisplay').textContent = name;
      showModal('Your Torch Is Lit', r.message || 'Your vote has been cast.');
      loadAppData();
    })
    .withFailureHandler(err => showModal('Submission Error', getErrorMessage(err)))
    .submitPicks(payload);
}

function renderEliteWeekPicks(data) {
  const container = document.getElementById('picksViewContainer');
  container.innerHTML = '';

  if (data.hidden) {
    container.innerHTML = '<div class="announcement">Picks will be revealed after the Wednesday 8:00 PM episode update.</div>';
    return;
  }

  if (!data.rows || !data.rows.length) {
    container.innerHTML = `<div class="announcement">No picks found for Week ${data.week}.</div>`;
    return;
  }

  const columns = data.columns || [];

  if (!columns.length) {
    container.innerHTML = `<div class="announcement">No used questions found for Week ${data.week}.</div>`;
    return;
  }

  const questionHeader = columns
    .map(col => `<th>${escapeHtml(col.label || col.key || '')}</th>`)
    .join('');

  const table = document.createElement('table');
  table.className = 'elite-picks-table';
  table.innerHTML = `
    <thead>
      <tr>
        <th>Rank</th>
        <th>Player</th>
        <th>Week Pts</th>
        <th>Total</th>
        ${questionHeader}
      </tr>
    </thead>
    <tbody></tbody>
  `;

  const tbody = table.querySelector('tbody');

  data.rows.forEach(row => {
    const tr = document.createElement('tr');
    tr.innerHTML = `
      <td>${escapeHtml(String(row.rank || ''))}</td>
      <td><strong>${escapeHtml(row.name || '')}</strong>${row.autoAssigned ? '<div class="tiny-note">Auto-assigned</div>' : ''}</td>
      <td>${escapeHtml(String(row.weekPoints ?? ''))}</td>
      <td>${escapeHtml(String(row.total ?? ''))}</td>
      ${columns.map(col => renderElitePickColumn_(row, col)).join('')}
    `;
    tbody.appendChild(tr);
  });

  const wrap = document.createElement('div');
  wrap.className = 'table-wrap';

  const thoughts = renderCampfireThoughts(data);
  if (thoughts) container.appendChild(thoughts);

  wrap.appendChild(table);
  container.appendChild(wrap);
}

function renderElitePickColumn_(row, col) {
  const picks = row.picks || {};
  return `<td>${renderPickCell_(picks[col.key], col.correct)}</td>`;
}

function renderCampfireThoughts(data) {
  const comments = (data.rows || [])
    .map(row => ({
      name: row.name || '',
      text: (row.picks && row.picks.comment) ? row.picks.comment : ''
    }))
    .filter(row => row.text.trim());

  if (!comments.length) return null;

  const section = document.createElement('div');
  section.className = 'campfire-thoughts';
  section.innerHTML = `
    <h3 class="section-title mini">${escapeHtml((data.headers && data.headers.comment) || 'Campfire Thoughts')}</h3>
    <div class="campfire-thoughts-grid"></div>
  `;

  const grid = section.querySelector('.campfire-thoughts-grid');
  comments.forEach(comment => {
    const card = document.createElement('div');
    card.className = 'campfire-thought-card';
    card.innerHTML = `
      <div class="campfire-thought-name">${escapeHtml(comment.name)}</div>
      <div class="campfire-thought-text">${escapeHtml(comment.text)}</div>
    `;
    grid.appendChild(card);
  });

  return section;
}

function renderPickCell_(value, correctList) {
  const safeValue = escapeHtml(value || '');
  const normalized = String(value || '').trim().toLowerCase();
  const correct = (correctList || []).map(v => String(v || '').trim().toLowerCase());

  if (!safeValue) return '<span class="pick-pill neutral">—</span>';
  if (!correct.length) return `<span class="pick-pill neutral">${safeValue}</span>`;
  if (correct.includes(normalized)) return `<span class="pick-pill correct">${safeValue}</span>`;
  return `<span class="pick-pill plain">${safeValue}</span>`;
}

function startCountdown(targetId, iso, endedText) {
  const el = document.getElementById(targetId);
  if (!el) return;
  if (!iso) {
    el.textContent = '--';
    return;
  }

  function tick() {
    const now = new Date();
    const target = new Date(iso);
    const diff = target.getTime() - now.getTime();

    if (diff <= 0) {
      el.textContent = endedText || 'Live';
      return;
    }

    const days = Math.floor(diff / 86400000);
    const hours = Math.floor((diff % 86400000) / 3600000);
    const mins = Math.floor((diff % 3600000) / 60000);
    const secs = Math.floor((diff % 60000) / 1000);

    const parts = [];
    if (days) parts.push(`${days}d`);
    parts.push(`${hours}h`);
    parts.push(`${mins}m`);
    parts.push(`${secs}s`);
    el.textContent = parts.join(' ');
  }

  tick();
  setInterval(tick, 1000);
}

function initDropZone() {
  const dropZone = document.getElementById('dropZone');
  const photoInput = document.getElementById('photoInput');

  if (!dropZone || !photoInput) return;

  dropZone.addEventListener('click', () => photoInput.click());

  photoInput.addEventListener('change', e => {
    const file = e.target.files && e.target.files[0];
    if (file) loadPhotoFile(file);
  });

  ['dragenter', 'dragover'].forEach(evt => {
    dropZone.addEventListener(evt, e => {
      e.preventDefault();
      dropZone.classList.add('dragover');
    });
  });

  ['dragleave', 'drop'].forEach(evt => {
    dropZone.addEventListener(evt, e => {
      e.preventDefault();
      dropZone.classList.remove('dragover');
    });
  });

  dropZone.addEventListener('drop', e => {
    const file = e.dataTransfer.files && e.dataTransfer.files[0];
    if (file) loadPhotoFile(file);
  });
}

function loadPhotoFile(file) {
  if (!file.type.startsWith('image/')) {
    showModal('Invalid File', 'Please upload an image file.');
    return;
  }

  PHOTO_FILE_NAME = file.name;
  PHOTO_MIME_TYPE = file.type;

  const reader = new FileReader();
  reader.onload = e => {
    PHOTO_DATA_URL = e.target.result || '';
    document.getElementById('photoPreview').src = PHOTO_DATA_URL;
    document.getElementById('photoPreviewWrap').classList.remove('hidden');
  };
  reader.readAsDataURL(file);
}

function uploadPhoto() {
  if (!APP_DATA) return;
  if (!ADMIN_PASSCODE) {
    showModal('Admin Only', 'Unlock the Admin Tribe Council before uploading game photos.');
    return;
  }
  const name = document.getElementById('photoName').value.trim();
  const email = document.getElementById('photoEmail').value.trim();
  const caption = document.getElementById('photoCaption').value.trim();

  if (!name) {
    showModal('Missing Name', 'Please enter your player name.');
    return;
  }
  if (!PHOTO_DATA_URL) {
    showModal('Missing Photo', 'Please choose a photo first.');
    return;
  }

  google.script.run
    .withSuccessHandler(r => {
      showModal('Snapshot Added', r.message || 'Photo uploaded.');
      document.getElementById('photoCaption').value = '';
      document.getElementById('photoPreviewWrap').classList.add('hidden');
      PHOTO_DATA_URL = '';
      PHOTO_FILE_NAME = '';
      PHOTO_MIME_TYPE = '';
      loadAppData();
    })
    .withFailureHandler(err => showModal('Upload Error', getErrorMessage(err)))
    .uploadTribePhoto(ADMIN_PASSCODE, {
      week: Number(document.getElementById('adminRecapWeek').value || APP_DATA.weekNumber),
      name,
      email,
      caption,
      dataUrl: PHOTO_DATA_URL,
      fileName: PHOTO_FILE_NAME,
      mimeType: PHOTO_MIME_TYPE
    });
}

function showModal(title, message) {
  document.getElementById('modalTitle').textContent = title || 'Notice';
  document.getElementById('modalBody').innerHTML = `<p>${escapeHtml(message || '')}</p>`;
  document.getElementById('parchmentModal').classList.remove('hidden');
}

function closeModal() {
  document.getElementById('parchmentModal').classList.add('hidden');
}

function getErrorMessage(err) {
  if (!err) return 'Unknown error';
  return err.message || String(err);
}

function escapeHtml(value) {
  return String(value || '')
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;');
}

function escapeAttr(value) {
  return escapeHtml(value);
}

function stripHtml(value) {
  const div = document.createElement('div');
  div.innerHTML = value || '';
  return (div.textContent || div.innerText || '').trim();
}
